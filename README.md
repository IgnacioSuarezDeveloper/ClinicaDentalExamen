class Tienda
{
    static void Main()
    {
        string nombreCliente = "Rosa";
        int tipoCliente = 2; // 1 = normal, 2 = premium
        double totalCompra = 0;

        Console.WriteLine("Introduce la cantidad de productos:");
        int cantidad = int.Parse(Console.ReadLine());

        for (int i = 0; i < cantidad; i++)
        {
            Console.WriteLine("Introduce el precio del producto " + (i + 1) + ":");
            double precio = double.Parse(Console.ReadLine());
            totalCompra = totalCompra + precio;
        }

        double descuento = 0;

        if (tipoCliente == 1)
        {
            if (totalCompra >= 100)
            {
                descuento = totalCompra * 0.10;
            }
            else
            {
                descuento = totalCompra * 0.05;
            }
        }
        else
        {
            if (tipoCliente == 2)
            {
                if (totalCompra >= 100)
                {
                    descuento = totalCompra * 0.20;
                }
                else
                {
                    descuento = totalCompra * 0.15;
                }
            }
        }

        double totalFinal = totalCompra - descuento;

        Console.WriteLine("Cliente: " + nombreCliente);
        Console.WriteLine("Tipo de cliente: " + tipoCliente);
        Console.WriteLine("Total compra: " + totalCompra);
        Console.WriteLine("Descuento: " + descuento);
        Console.WriteLine("Total final: " + totalFinal);
    }
}

# BarCopa

## Descripción
Se desarrolló una página estática para el Clinica dental
La página contiene una cabecera con su titulo y un enlace a Inicio, Servicios, Citas, Equipo Medico  
También incluye un contenido principal donde se muestra un breve resumen de los servicios ofrecidos con su descripcion,
y los horarios
## Tecnología

Proyecto de una página web estática creada con **HTML y CSS**.

## Estructura

ClinicaDental/
│
├── index.html
├── style.css


## Cómo abrir el proyecto

1. Descargar la carpeta del proyecto.
2. Abrir el archivo **index.html** en el navegador.
